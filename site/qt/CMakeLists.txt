cmake_minimum_required(VERSION 3.25)
project(FreeFlightSite VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.12 REQUIRED COMPONENTS Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.12)
if(NOT EXISTS "${FREEFLIGHT_SITE_CONTENT}")
  message(FATAL_ERROR "Generate FreeFlight content with the DockPipe site.build workflow")
endif()
qt_add_executable(freeflight-site main.cpp)
configure_file(site-config.h.in "${CMAKE_CURRENT_BINARY_DIR}/site-config.h" @ONLY)
target_include_directories(freeflight-site PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(freeflight-site PROPERTIES QT_WASM_EXPORT_NAME freeflight_site_entry)
qt_add_qml_module(freeflight-site URI FreeFlight.Site VERSION 1.0 QML_FILES Main.qml)
set_source_files_properties("${FREEFLIGHT_SITE_CONTENT}" PROPERTIES QT_RESOURCE_ALIAS content.json)
qt_add_resources(freeflight-site content PREFIX / FILES "${FREEFLIGHT_SITE_CONTENT}")
qt_add_resources(freeflight-site fonts PREFIX / FILES fonts/InterVariable.ttf fonts/DejaVuSansMono.ttf)
target_link_libraries(freeflight-site PRIVATE Qt6::Quick Qt6::QuickControls2)
if(EMSCRIPTEN)
  configure_file(shell.html "${CMAKE_CURRENT_BINARY_DIR}/index.html" COPYONLY)
  configure_file(loader.js "${CMAKE_CURRENT_BINARY_DIR}/loader.js" COPYONLY)
endif()
