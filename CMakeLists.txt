cmake_minimum_required(VERSION 3.25)
project(FreeFlight VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json
  URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(json)
find_package(Threads REQUIRED)
add_executable(freeflight src/main.cpp src/server.cpp src/process.cpp)
target_include_directories(freeflight PRIVATE include)
target_link_libraries(freeflight PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
if(MSVC)
  target_compile_options(freeflight PRIVATE /W4 /permissive-)
else()
  target_compile_options(freeflight PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
find_package(Python3 COMPONENTS Interpreter)
if(BUILD_TESTING AND Python3_FOUND)
  add_test(NAME protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/protocol.py $<TARGET_FILE:freeflight>)
endif()
install(TARGETS freeflight RUNTIME DESTINATION bin)
install(FILES LICENSE README.md DESTINATION share/freeflight)
install(DIRECTORY docs/ DESTINATION share/freeflight/docs)
install(DIRECTORY site/ DESTINATION share/freeflight/site OPTIONAL)
set(CPACK_PACKAGE_NAME freeflight)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR TGZ)
include(CPack)

if(BUILD_TESTING)
  add_executable(freeflight_process_tests tests/process.cpp src/process.cpp)
  target_include_directories(freeflight_process_tests PRIVATE include)
  target_link_libraries(freeflight_process_tests PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
  if(WIN32)
    target_link_libraries(freeflight_process_tests PRIVATE shell32)
  endif()
  if(MSVC)
    target_compile_options(freeflight_process_tests PRIVATE /utf-8)
  endif()
  add_test(NAME process COMMAND freeflight_process_tests)
  set_tests_properties(process PROPERTIES TIMEOUT 60)
endif()
